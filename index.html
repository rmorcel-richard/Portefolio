const SHEET_NAME = 'Donnees'; const HEADERS = [ 'date', 'type_evenement', 'page', 'appareil', 'ville', 'pays', 'temps_reste', 'latitude', 'longitude' ]; function doGet() { try { const sheet = getDataSheet_(); ensureHeaders_(sheet); const values = sheet.getDataRange().getDisplayValues(); if (values.length < 2) { return jsonResponse_({ success: true, rows: [] }); } const headers = values[0].map(normalizeHeader_); const rows = values.slice(1) .filter(row => row.some(cell => String(cell).trim() !== '')) .map(row => { const item = {}; headers.forEach((header, index) => { item[header] = row[index] ?? ''; }); return item; }); return jsonResponse_({ success: true, rows: rows }); } catch (error) { return jsonResponse_({ success: false, error: String(error && error.message ? error.message : error), rows: [] }); } } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const sheet = getDataSheet_(); ensureHeaders_(sheet); let data = {}; if (e && e.postData && e.postData.contents) { try { data = JSON.parse(e.postData.contents); } catch (jsonError) { data = e.parameter || {}; } } else if (e && e.parameter) { data = e.parameter; } const dateValue = data.date_visite || data.date || Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm:ss'); sheet.appendRow([ dateValue, data.type_evenement || '', data.page || '', data.appareil || '', data.ville || '', data.pays || '', data.temps_reste || '', data.latitude || '', data.longitude || '' ]); return jsonResponse_({ success: true }); } catch (error) { return jsonResponse_({ success: false, error: String(error && error.message ? error.message : error) }); } finally { try { lock.releaseLock(); } catch (_) {} } } function getDataSheet_() { const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); if (!spreadsheet) { throw new Error('Aucun Google Sheets associé à ce script.'); } const sheet = spreadsheet.getSheetByName(SHEET_NAME); if (!sheet) { throw new Error('Onglet introuvable : ' + SHEET_NAME); } return sheet; } function ensureHeaders_(sheet) { const current = sheet.getRange(1, 1, 1, HEADERS.length).getDisplayValues()[0]; const normalized = current.map(normalizeHeader_); const correct = HEADERS.every((header, index) => normalized[index] === header); if (correct) return; const rowIsEmpty = current.every(value => String(value).trim() === ''); if (rowIsEmpty) { sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); sheet.setFrozenRows(1); return; } // Migration automatique depuis l'ancienne version à 7 colonnes. const legacyHeaders = HEADERS.slice(0, 7); const legacyCorrect = legacyHeaders.every((header, index) => normalized[index] === header); const newColumnsEmpty = current.slice(7).every(value => String(value).trim() === ''); if (legacyCorrect && newColumnsEmpty) { sheet.getRange(1, 8, 1, 2).setValues([['latitude', 'longitude']]); sheet.setFrozenRows(1); return; } throw new Error( 'Les colonnes de la ligne 1 doivent être exactement : ' + HEADERS.join(' | ') ); } function normalizeHeader_(value) { return String(value || '') .trim() .toLowerCase() .normalize('NFD') .replace(/[\u0300-\u036f]/g, '') .replace(/[^a-z0-9]+/g, '_') .replace(/^_+|_+$/g, ''); } function jsonResponse_(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
